\begin{frame}{Methodology}{Datasets and Feature Extraction}
  \begin{itemize}
    \item \textbf{NIH Chest X-ray 10k}: 7,500 SAE training / 2,500 validation images; no text or labels.
    \item \textbf{Open-I}: 3,666 frontal image--report pairs form the semantic evaluation. We combine Findings and Impression into reference reports to check extracted concepts.
    \item \textbf{Features}: Frozen BiomedCLIP ViT-B/16 image encoder; \textbf{512-dimensional embeddings}.
  \end{itemize}
  {\centering
    \includegraphics[width=0.60\textwidth,keepaspectratio]{assets/Biomedclip.png}
    \par
    {\footnotesize BiomedCLIP overview\par}
  }

  \note{
    \begin{itemize}
      \item \textbf{Two datasets, different roles}
      \item \textbf{NIH Chest X-ray}: To train the SAE. Train only on image embeddings, without disease labels or reports.
      \item \textbf{Open-I}: Keep frontal images. Combine \textbf{Findings} (radiological observations) and \textbf{Impression} (diagnostic interpretation) into a reference report to check whether the extracted concepts were supported.
      \item \textbf{Frozen BiomedCLIP image encoder}: Convert each X-ray into a \textbf{512-dimensional embedding}, the input to the SAE.
    \end{itemize}
  }
\end{frame}


\begin{frame}{Methodology}{Concept Dictionary Construction}
  \begin{itemize}
    \item \textbf{Concept Selection}: 100 parent concepts from cleaned and filtered Open-I Problems metadata, covering anatomy and pathological findings.
    \item \textbf{Expansion}: UMLS provides Concept Unique Identifiers (CUIs) and synonyms from SNOMED CT and MeSH for matched concepts.
    \item \textbf{Phrase Generation}: Gemini 3.1 Flash-Lite generates 5 radiology-style descriptions per concept.
    \item \textbf{Encoding}: BiomedCLIP's PubMedBERT text encoder maps all 998 dictionary entries to L2-normalized, 512-dimensional embeddings in the same space as the images.
  \end{itemize}
  {\Large
  \[
    T \in \mathbb{R}^{998 \times 512}
  \]
  }

  \note{
    \begin{itemize}
      \item \textbf{To name the features}: learned by the SAE, we build a \textbf{dictionary of medical concepts}.
      \item \textbf{100 parent concepts}: From Open-I \textbf{Problems} metadata, filter devices, technical terms. Retain \textbf{anatomical structures and pathological findings}.
      \item \textbf{UMLS expansion}: Retrieve CUIs and synonyms from medical vocabularies. Keep the original term when no match exists.
      \item \textbf{Gemini 3.1 Flash-Lite}: Generate \textbf{five short radiology phrases per concept} in language of radiology reports, describing consistent visible features.
      \item \textbf{PubMedBERT text encoder}: Encode 498 terms/synonyms and 500 generated phrases. Matrix $T$ contains \textbf{998 embeddings of 512 dimensions}, in the same space as the images.
    \end{itemize}
  }
\end{frame}


\begin{frame}{Methodology}{Sparse Autoencoder Architecture and Training}
  \begin{columns}[c,onlytextwidth]
  \begin{column}{0.66\textwidth}
  \begin{itemize}
    \item \textbf{Architecture}: 512-dimensional input, 1,024 hidden units
    (overcomplete factor of 2), ReLU activations $z$.
    \medskip
    \item \textbf{Objective}: Reconstruction error plus an $L_1$ penalty on activations; $\lambda$ controls the sparsity penalty.
    \[
      \mathcal{L}=\frac{1}{512B}\sum_{i=1}^{B}\|x_i-\hat{x}_i\|_2^2
      +\frac{\lambda}{B}\sum_{i=1}^{B}\|z_i\|_1
    \]
    \item \textbf{Optimization}: Adam, batch size $B=256$. Validation loss guides early stopping and model selection.
  \end{itemize}
  \end{column}
  \begin{column}{0.30\textwidth}
    \centering
    \includegraphics[width=\linewidth,keepaspectratio]{assets/SAE_diagram.png}
    \par\smallskip
    {\footnotesize Illustrative SAE architecture}
  \end{column}
  \end{columns}

  \note{
    \begin{itemize}
      \item \textbf{Goal}: Represent each image embedding as a \textbf{sparse combination of learned features}.
      \item \textbf{Architecture}: 512 input dimensions, 1,024 hidden units and ReLU activations. Overcomplete factor of \textbf{two}, as illustrated on the right.
      \item \textbf{Reconstruction loss}: Preserve information from the input embedding.
      \item \textbf{$L_1$ penalty}: Encourage sparse activations. $\lambda$ controls the penalty strength.
      \item \textbf{Training}: Adam, batch size 256. Use \textbf{validation loss} for early stopping and model selection.
    \end{itemize}
  }
\end{frame}


\begin{frame}{Methodology}{Concept Assignment and Inference}
  After SAE training, each decoder column represents a learned feature direction
  in the image embedding space.
  \medskip
  \begin{itemize}
    \item \textbf{Naming}: Compare normalized decoder columns $\widetilde{W}_d$ with
    dictionary embeddings $T$. Assign the entry with the highest cosine similarity
    to each feature.
    {
    \[
      S=T\widetilde{W}_d\in\mathbb{R}^{998\times1024}
    \]
    }
    \item \textbf{Inference}: Rank positive SAE activations for each Open-I image.
    Select features from largest to smallest until cumulative activation $>$ \textbf{90\% of the total positive activation}.
    \item \textbf{Output}: Collect the selected features' labels and remove duplicate
    phrases to obtain the predicted concept set.
  \end{itemize}

  \note{
    \begin{itemize}
      \item After training, each \textbf{Decoder columns} represents a \textbf{learned feature direction} in the image embedding space.
      \item To name \textbf{Features}, compare normalized decoder columns with dictionary embeddings $T$ using \textbf{cosine similarity}, producing matrix $S$.
      \item Assign each feature the \textbf{highest-similarity dictionary entry}.
      \item \textbf{Image inference}: Rank positive SAE activations from largest to smallest. Select features until cumulative activation \textbf{exceeds 90\%} of the total positive activation.
      \item \textbf{Predicted concept set}: Collect labels and remove duplicates. The number of selected features varies across images.
    \end{itemize}
  }
\end{frame}


\begin{frame}{Methodology}{Semantic Evaluation and Validation}
  \begin{columns}[c,onlytextwidth]
  \begin{column}{0.60\textwidth}
  \begin{itemize}
    \item \textbf{Evaluator}: Frozen MedGemma 1.5 4B;
    concept and report only (No image).
    \item \textbf{Verdicts}:
    \begin{itemize}
        \item[{\makebox[1em][c]{\textcolor{green!60!black}{$\checkmark$}}}] \textbf{Aligned}: Explicit support.
        \item[{\makebox[1em][c]{\textcolor{red}{$\times$}}}] \textbf{Unaligned}: Explicit contradiction.
        \item[{\makebox[1em][c]{\textcolor{orange}{$?$}}}] \textbf{Uncertain}: Unmentioned, ambiguous, or tentative.
    \end{itemize}
    \item \textbf{Clustering}: Hierarchical parent-concept groups;
    To compare alignment across clusters.
    \item \textbf{Validation}: 25 concept-report pairs, 3 independet authors.
    Majority vote as reference for MedGemma and 3 commercial models.
  \end{itemize}
  \end{column}
  \begin{column}{0.37\textwidth}
    \centering
    \includegraphics[width=\linewidth,height=5.5cm,keepaspectratio]{assets/Evaluation.png}
  \end{column}
  \end{columns}

  \note{
    \begin{itemize}
      \item To \textbf{evaluate} whether the extracted concepts are \textbf{supported} by the reference report. \textbf{Frozen MedGemma 1.5}, receives concept phrase and report, \textbf{without the image}.
      \item \textbf{Verdicts}: Aligned = explicit support. Unaligned = contradiction or negation. Uncertain = unmentioned or ambiguous.
      \item \textbf{Reduce letter bias}: Run three forward passes, rotating the A/B/C verdict mapping. Average verdict probabilities.
      \item \textbf{Hierarchical clustering}: Group related parent concepts and compare alignment across groups, reusing existing verdicts.
      \item \textbf{To check the judge's reliability}: Independently label \textbf{25 concept--report pairs}, blinded to MedGemma verdicts. Use author majority vote as the reference for MedGemma and \textbf{three commercial models}.
    \end{itemize}
  }
\end{frame}
