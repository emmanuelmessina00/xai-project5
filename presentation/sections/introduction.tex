% --- Frame 1: The Problem ---
\begin{frame}{Introduction}{Why Are Medical VLMs Opaque?}
  \begin{itemize}
    \item Medical \textbf{Vision-Language Models} (VLMs) encode images and text in a shared representation with strong clinical performance
    \item But their latent representations are \textbf{high-dimensional} and \textbf{polysemantic}
    \begin{itemize}
      \item Each dimension mixes multiple clinical meanings
    \end{itemize}
    \item This limits \textbf{interpretability} and \textbf{trust} in safety-critical applications
    \item \textbf{Concept-based explanations} aim to map representations onto recognizable anatomy and findings
  \end{itemize}

  \note{
    Medical VLMs like BiomedCLIP encode images and text in a shared space with excellent performance.
    However, their latent representations are polysemantic: each dimension mixes multiple clinical meanings, making it hard to understand what the model has learned.
    This limits interpretability and trustworthiness, which are critical in safety-sensitive applications like medicine.
    Concept-based explanations try to solve this by mapping these opaque representations onto recognizable clinical concepts.
  }
\end{frame}

% --- Frame 2: Our Approach ---
\begin{frame}{Introduction}{Our Approach}
  \begin{itemize}
    \item Following \textbf{MedConcept}: extract SAE features \textbf{without diagnostic supervision}, name them, and evaluate against radiology reports
    \item \textbf{Adaptation} from the original pipeline:
    \begin{itemize}
      \item 3D abdominal CT $\rightarrow$ \textbf{2D chest radiographs}
      \item Backbone: \textbf{BiomedCLIP} (frozen ViT-B/16)
      \item New vocabulary and activation selection
    \end{itemize}
    \item \textbf{Objective}: Assess what the evidence supports about interpretability --- \textit{not} a diagnostic classifier
  \end{itemize}

  \note{
    Following MedConcept, we use a Sparse Autoencoder to decompose embeddings into sparse latent features without disease labels, and validate them against actual radiology reports.
    We adapted the original pipeline by swapping the backbone to BiomedCLIP and building a new dedicated concept vocabulary for 2D chest X-rays.
    Important: our goal is not to build a classifier, but to assess what the evidence actually tells us about interpretability.
  }
\end{frame}
