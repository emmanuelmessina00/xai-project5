% --- Frame 1: The Four Research Gaps ---
\begin{frame}{Research Gaps}{Four Gaps Guide Our Analysis}
  \begin{enumerate}
    \item \textbf{Reconstruction $\neq$ Correct Naming}\\
    Vocabulary coverage and image--text geometry constrain naming

    \item \textbf{Dependence on External Resources}\\
    Ontology, generative phrasing, LLM judges --- errors can be mistaken for representation properties

    \item \textbf{Reports as Selective Evidence}\\
    Omission $\neq$ absence --- judges can confuse the two

    \item \textbf{Independent Labels Omit Relationships}\\
    Embedding similarity $\neq$ anatomical hierarchy
  \end{enumerate}

  \note{
    Four gaps guide our analysis.
    First: good reconstruction does not mean correct naming. The vocabulary and the embedding geometry constrain what labels are even possible.
    Second: the pipeline depends on external resources at every stage. Errors in these tools can look like properties of the representation itself.
    Third: reports are partial. Radiologists omit normal anatomy, and the judge can confuse omission with contradiction.
    Fourth: assigning labels independently ignores relationships between concepts.
  }
\end{frame}

% --- Frame 2: Our Response to the Gaps ---
\begin{frame}{Research Gaps}{Our Response}
  \begin{itemize}
    \item \textbf{Hierarchical clustering} at multiple resolutions to examine concept groups
    \item \textbf{Human + multi-model audit} on 25 stratified pairs to assess \textbf{evaluator reliability}
    \begin{itemize}
      \item 3 human annotators + 3 commercial LLMs
    \end{itemize}
    \item \textbf{Label permutation} to mitigate position bias in the LLM judge
  \end{itemize}

  \vspace{0.4cm}
  \centering
  \textit{$\rightarrow$ Goal: assess interpretability in a scalable and reproducible way}

  \note{
    To address these gaps: we use hierarchical clustering to examine concept groups, a blinded audit with three annotators and three commercial LLMs to measure evaluator reliability, and label permutation to mitigate judge bias.
    The broader goal is assessing interpretability in a way that is scalable and reproducible.
  }
\end{frame}
