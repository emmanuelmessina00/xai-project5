% --- Frame 1: From Concepts to Medical Grounding ---
\begin{frame}{Related Work}{From Predefined Concepts to Unsupervised Discovery}
  \begin{itemize}
    \item \textbf{Supervised concept methods} (TCAV, Concept Bottleneck Models): semantic interfaces, but require \textbf{predefined annotations}
    \item \textbf{Sparse dictionary learning}: decomposes polysemantic activations into monosemantic features \textit{without} predefined concepts
    \item Recent approaches go further:
    \begin{itemize}
      \item First \textit{discover} features, then \textit{name} them --- decoupling representation from labeling
      \item Decompose VLM embeddings directly into sparse combinations of existing textual concepts
    \end{itemize}
    \item In the \textbf{medical domain}: recent works dissect neuron activations in diagnostic models to extract concept-level explanations from opaque networks
  \end{itemize}

  \note{
    Traditional methods like TCAV and Concept Bottleneck Models need predefined annotations.
    Sparse dictionary learning removes this by extracting monosemantic features unsupervisedly.
    Recent approaches decouple discovery from naming, or decompose VLM embeddings directly into textual concepts.
    In the medical domain, several works extract interpretable concepts from opaque diagnostic networks, which motivates our approach.
  }
\end{frame}
